In this section, we list additional pseudocode for the \sys protocols described in \autoref{sec:protocol}.
\autoref{alg:membership} shows the pseudocode for membership maintenance as introduced in \autoref{sec:protocol:detail:group}.
\autoref{alg:repair} includes procedures for repairing chunk fragments as discussed in \autoref{sec:protocol:detail:repair}.

\pagebreak

\begin{algorithm}[t]
    \begin{algorithmic}[1]
        \Procedure{MembershipTimer}{}
            \State $\fn{UpdateGroup}$
            \State $\fn{RepairChunk}$
        \EndProcedure
        \Procedure{HeartbeatTimer}{}
            \State $\var{fhash} \gets \fn{Hash}(\var{frag})$
            \State $\langle r, \pi, \_ \rangle \gets \fn{SelectionProof}(\var{fhash})$
            \ForAll{$\var{n} \in \var{group}$}
                \State $\fn{Send}(\msg{HeartBeat}{\var{index, \langle r, \pi \rangle}})$
            \EndFor
        \EndProcedure

        \Procedure{RecvHeartBeat}{$i, \langle r, \pi \rangle$}
            \If{$i \notin \set{\wl .. \wh}$}
                \State \Return
            \EndIf
            \State $\var{fhash} \gets \fn{InnerEncodeHash}(\var{chash}, i)$
            \If{$\fn{VerifySelection}(\var{fhash}, r, \pi, \var{pk}_{sender})$}
                \State insert sender into $\var{group}$
                \State $\fn{UpdateGroup}$
            \EndIf
        \EndProcedure
        
        \Procedure{UpdateGroup}{}
            \State $\var{group'} \gets \set{}, i \gets \wl$
            \While{$i \leq \wh$}
                \State $\var{nodes_i} \gets \set{n \in \var{group} \mid n\textrm{'s fragment index} = i}$
                \State $\var{fhash_i} \gets \fn{InnerEncodeHash}(\var{chash}, i)$
                \State increment $i$
                \If{$n \in \var{nodes_i}$ is alive and closest to $\var{fhash_i}$}
                    \State insert $n$ into $\var{group'}$
                \EndIf
                \If{$|\var{group'}| = R$}
                    \State break
                \EndIf
            \EndWhile
            \State $\var{group} \gets \var{group'}$
            \State $\wh \gets i - 1$
        \EndProcedure
        \Procedure{UpdateWaterLTimer}{}
            \State increment $\wl$
        \EndProcedure
    \end{algorithmic}
    \caption{Procedures to maintain group membership}
    \label{alg:membership}
\end{algorithm}

\begin{algorithm}
    \begin{algorithmic}[1]
        \Procedure{RepairChunk}{}
            \State $i \gets \wl, \var{nr} \gets 0$
            \ForAll{$i \in \set{\wl .. \wh}$}
                \If{$\nexists n \in \var{group}: n.\var{index} = i$}
                    \If{$\fn{RepairFrag}(i)$}
                        \State increment $\var{nr}$
                    \EndIf
                \EndIf
            \EndFor
            \While{$\var{nr} < R - |group|$}
                \State increment $\wh$
                \If{$\fn{RepairFrag}(\wh)$}
                    \State increment $\var{nr}$
                \EndIf
            \EndWhile
            \If{not holding $\var{frag}$}
                \State $\var{rfrags} \gets \set{}$
                \While{$|\var{rfrags} < \kin|$}
                    \State wait $\msg{RepairResponse}{frag, \langle r, \pi \rangle}$
                    \State $\var{fh} \gets \fn{Hash}(frag)$
                    \If{$\fn{VerifySelection}(\var{fh}, r, \pi, \var{pk}_{sender})$}
                        \State insert $\var{frag}$ into $\var{rfrags}$
                        \State insert sender into $\var{group}$
                    \EndIf
                \EndWhile
                \State $\var{frag} \gets \fn{InnerDecode}(rfrags)$
            \EndIf
        \EndProcedure
        \Procedure{RepairFrag}{$\var{i}$}
            \State $\var{fhash} \gets \fn{InnerEncodeHash}(\var{chash}, i)$
            \State $\var{nodes} \gets \fn{Locate}(\var{fhash})$
            \ForAll{$n \in \var{nodes}$}
                \State send $\msg{RepairRequest}{i}$ to $n$
            \EndFor
            \State \Return $\var{nodes} \ne \set{}$
        \EndProcedure
        \Procedure{RecvRepairRequest}{$\var{i}$}
            \If{holding $\var{frag}$ for $\var{chash}$}
                \State $\var{fhash} \gets \fn{Hash}(\var{frag})$
                \State $\langle r, \pi, \_ \rangle \gets \fn{SelectionProof}(\var{fhash})$
                \State respond with $\msg{RepairResponse}{frag, \langle r, \pi \rangle}$
            \Else
                \State start $\fn{MembershipTimer}$ if not yet started
            \EndIf
        \EndProcedure
    \end{algorithmic}
    \caption{Protocol to repair encoded chunk}
    \label{alg:repair}
\end{algorithm}
